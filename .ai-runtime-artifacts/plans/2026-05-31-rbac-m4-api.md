# RBAC — M4: Permission Service & Admin API 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现 Role 和 Permission 的 CRUD service 层；新增 Admin RBAC 管理 API 路由（角色列表、权限分配、用户角色切换）。

**Architecture:** Service 层封装 GORM 操作；API 层使用 M3 的 `RequirePermission()` 中间件保护写操作。Role 本身可被任何人读取，编辑/删除需要 `roles:write`。

**Tech Stack:** Go（Gin）、GORM

- [ ] **Step 1: 创建 Role Service**

```go
package services

import (
    "apps/backend/models"
    "errors"

    "gorm.io/gorm"
)

var (
    ErrRoleNotFound    = errors.New("role not found")
    ErrSystemRoleDelete = errors.New("cannot delete system role")
)

type RoleService struct {
    db *gorm.DB
}

func NewRoleService(db *gorm.DB) *RoleService {
    return &RoleService{db: db}
}

func (s *RoleService) ListRoles() ([]models.Role, error) {
    var roles []models.Role
    err := s.db.Preload("Permissions").Find(&roles).Error
    return roles, err
}

func (s *RoleService) GetRole(id int64) (*models.Role, error) {
    var role models.Role
    err := s.db.Preload("Permissions").First(&role, id).Error
    if errors.Is(err, gorm.ErrRecordNotFound) {
        return nil, ErrRoleNotFound
    }
    return &role, err
}

func (s *RoleService) CreateRole(name, description string, isSystem bool) (*models.Role, error) {
    role := &models.Role{
        Name:        name,
        Description: description,
        IsSystem:    isSystem,
    }
    err := s.db.Create(role).Error
    return role, err
}

func (s *RoleService) UpdateRole(id int64, name, description string) (*models.Role, error) {
    role, err := s.GetRole(id)
    if err != nil {
        return nil, err
    }
    if name != "" {
        role.Name = name
    }
    if description != "" {
        role.Description = description
    }
    // Name of system roles cannot be changed
    err = s.db.Save(role).Error
    return role, err
}

func (s *RoleService) DeleteRole(id int64) error {
    role, err := s.GetRole(id)
    if err != nil {
        return err
    }
    if role.IsSystem {
        return ErrSystemRoleDelete
    }
    return s.db.Delete(role).Error
}

func (s *RoleService) SetRolePermissions(roleID int64, permissionIDs []int64) error {
    var role models.Role
    if err := s.db.First(&role, roleID).Error; err != nil {
        return ErrRoleNotFound
    }

    var permissions []models.Permission
    if len(permissionIDs) > 0 {
        s.db.Find(&permissions, permissionIDs)
    }

    return s.db.Model(&role).Association("Permissions").Replace(permissions)
}

func (s *RoleService) GetUserRole(userID int64) (*models.Role, error) {
    var user models.User
    err := s.db.First(&user, userID).Error
    if err != nil {
        return nil, err
    }
    if user.RoleID == nil {
        return nil, nil
    }
    return s.GetRole(*user.RoleID)
}
```

package services

type PermissionService struct {
    db *gorm.DB
}

func NewPermissionService(db *gorm.DB) *PermissionService {
    return &PermissionService{db: db}
}

func (s *PermissionService) ListPermissions(category string) ([]models.Permission, error) {
    var permissions []models.Permission
    query := s.db
    if category != "" {
        query = query.Where("category = ?", category)
    }
    err := query.Order("category, code").Find(&permissions).Error
    return permissions, err
}

func (s *PermissionService) GetPermission(id int64) (*models.Permission, error) {
    var p models.Permission
    err := s.db.First(&p, id).Error
    if errors.Is(err, gorm.ErrRecordNotFound) {
        return nil, errors.New("permission not found")
    }
    return &p, err
}

## Task 4: 创建 Admin RBAC API 路由

### 4.4 创建 RBAC Admin 路由

**Files:**
- Create: `apps/backend/routes/admin_rbac.go`

- [ ] **Step 1: 创建 Admin RBAC 路由文件**

```go
package routes

import (
    "apps/backend/internal/middleware"
    "apps/backend/services"
    "strconv"

    "github.com/gin-gonic/gin"
)

func RegisterRBACRoutes(r *gin.RouterGroup, db *gorm.DB) {
    roleSvc := services.NewRoleService(db)
    permSvc := services.NewPermissionService(db)
    userSvc := services.NewUserService(db)
    rbacMw := middleware.NewRBACMiddleware(db)

    // Load permissions for all RBAC routes
    r.Use(rbacMw.LoadUserPermissions())

    // Permissions — read only (no write protection needed for listing)
    r.GET("/permissions", func(c *gin.Context) {
        category := c.Query("category")
        perms, err := permSvc.ListPermissions(category)
        if err != nil {
            c.JSON(500, gin.H{"error": err.Error()})
            return
        }
        c.JSON(200, gin.H{"success": true, "data": perms})
    })

    // Roles
    r.GET("/roles", middleware.RequirePermission("roles:read"), func(c *gin.Context) {
        roles, err := roleSvc.ListRoles()
        if err != nil {
            c.JSON(500, gin.H{"error": err.Error()})
            return
        }
        c.JSON(200, gin.H{"success": true, "data": roles})
    })

    r.POST("/roles", middleware.RequirePermission("roles:write"), func(c *gin.Context) {
        var body struct {
            Name        string `json:"name" binding:"required"`
            Description string `json:"description"`
        }
        if err := c.ShouldBindJSON(&body); err != nil {
            c.JSON(400, gin.H{"error": err.Error()})
            return
        }
        role, err := roleSvc.CreateRole(body.Name, body.Description, false)
        if err != nil {
            c.JSON(500, gin.H{"error": err.Error()})
            return
        }
        c.JSON(200, gin.H{"success": true, "data": role})
    })

    r.GET("/roles/:id", middleware.RequirePermission("roles:read"), func(c *gin.Context) {
        id, _ := strconv.ParseInt(c.Param("id"), 10, 64)
        role, err := roleSvc.GetRole(id)
        if err != nil {
            c.JSON(404, gin.H{"error": err.Error()})
            return
        }
        c.JSON(200, gin.H{"success": true, "data": role})
    })

    r.PUT("/roles/:id", middleware.RequirePermission("roles:write"), func(c *gin.Context) {
        id, _ := strconv.ParseInt(c.Param("id"), 10, 64)
        var body struct {
            Name        string `json:"name"`
            Description string `json:"description"`
        }
        if err := c.ShouldBindJSON(&body); err != nil {
            c.JSON(400, gin.H{"error": err.Error()})
            return
        }
        role, err := roleSvc.UpdateRole(id, body.Name, body.Description)
        if err != nil {
            c.JSON(404, gin.H{"error": err.Error()})
            return
        }
        c.JSON(200, gin.H{"success": true, "data": role})
    })

    r.DELETE("/roles/:id", middleware.RequirePermission("roles:write"), func(c *gin.Context) {
        id, _ := strconv.ParseInt(c.Param("id"), 10, 64)
        if err := roleSvc.DeleteRole(id); err != nil {
            status := 500
            if err == services.ErrSystemRoleDelete || err == services.ErrRoleNotFound {
                status = 400
            }
            c.JSON(status, gin.H{"error": err.Error()})
            return
        }
        c.JSON(200, gin.H{"success": true})
    })

    // Role permissions
    r.GET("/roles/:id/permissions", middleware.RequirePermission("roles:read"), func(c *gin.Context) {
        id, _ := strconv.ParseInt(c.Param("id"), 10, 64)
        role, err := roleSvc.GetRole(id)
        if err != nil {
            c.JSON(404, gin.H{"error": err.Error()})
            return
        }
        c.JSON(200, gin.H{"success": true, "data": role.Permissions})
    })

    r.PUT("/roles/:id/permissions", middleware.RequirePermission("roles:write"), func(c *gin.Context) {
        id, _ := strconv.ParseInt(c.Param("id"), 10, 64)
        var body struct {
            PermissionIDs []int64 `json:"permission_ids"`
        }
        if err := c.ShouldBindJSON(&body); err != nil {
            c.JSON(400, gin.H{"error": err.Error()})
            return
        }
        if err := roleSvc.SetRolePermissions(id, body.PermissionIDs); err != nil {
            c.JSON(500, gin.H{"error": err.Error()})
            return
        }
        role, _ := roleSvc.GetRole(id)
        c.JSON(200, gin.H{"success": true, "data": role})
    })

    // Assign role to user
    r.PUT("/users/:id/role", middleware.RequirePermission("users:write"), func(c *gin.Context) {
        userID, _ := strconv.ParseInt(c.Param("id"), 10, 64)
        var body struct {
            RoleID *int64 `json:"role_id"` // nil means remove role
        }
        if err := c.ShouldBindJSON(&body); err != nil {
            c.JSON(400, gin.H{"error": err.Error()})
            return
        }
        if err := userSvc.AssignRole(userID, body.RoleID); err != nil {
            c.JSON(500, gin.H{"error": err.Error()})
            return
        }
        c.JSON(200, gin.H{"success": true})
    })

    // Current user role & permissions
    r.GET("/me/permissions", func(c *gin.Context) {
        perms, exists := c.Get("permissions")
        if !exists {
            c.JSON(200, gin.H{"success": true, "data": []string{}})
            return
        }
        c.JSON(200, gin.H{"success": true, "data": perms})
    })
}
```
