# RBAC — M2: Permission 常量 & Seed 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 定义所有 Permission.code 常量；实现内置角色和权限的种子数据初始化；创建前端 TypeScript 类型。

**Architecture:** Go 端用 const 定义 permission code 字符串常量，seed 时插入 DB；前端用 TypeScript 类型生成同款常量。Seed 通过 GORM 的 AfterCreate hook 或独立 service 调用实现。

**Tech Stack:** Go、GORM、TypeScript

## Task 3: Seed Service

### 2.3 创建 Seed Service

**Files:**
- Create: `apps/backend/internal/rbac/seed.go`

- [ ] **Step 1: 创建 seed service**

```go
package rbac

import (
    "apps/backend/models"
    "gorm.io/gorm"
)

type SeedService struct {
    db *gorm.DB
}

func NewSeedService(db *gorm.DB) *SeedService {
    return &SeedService{db: db}
}

// Seed inserts system roles and all permissions if they don't exist
func (s *SeedService) Seed() error {
    return s.db.Transaction(func(tx *gorm.DB) error {
        if err := s.seedPermissions(tx); err != nil {
            return err
        }
        return s.seedRoles(tx)
    })
}

func (s *SeedService) seedPermissions(tx *gorm.DB) error {
    permissionDefs := []struct {
        Code        string
        Name        string
        Category    string
        Description string
    }{
        // Admin
        {PermTasksRead, "查看任务", CategoryAdmin, "查看所有任务列表"},
        {PermTasksWrite, "管理任务", CategoryAdmin, "创建和编辑任务"},
        {PermTasksDelete, "删除任务", CategoryAdmin, "删除任务"},
        {PermCollectionsRead, "查看集合", CategoryAdmin, "查看所有集合"},
        {PermCollectionsWrite, "管理集合", CategoryAdmin, "创建和编辑集合"},
        {PermCollectionsDelete, "删除集合", CategoryAdmin, "删除集合"},
        {PermUsersRead, "查看用户", CategoryAdmin, "查看用户列表"},
        {PermUsersWrite, "管理用户", CategoryAdmin, "创建和编辑用户"},
        {PermUsersDelete, "删除用户", CategoryAdmin, "删除用户"},
        {PermRolesRead, "查看角色", CategoryAdmin, "查看角色列表"},
        {PermRolesWrite, "管理角色", CategoryAdmin, "创建、编辑、删除角色"},
        {PermStatsView, "查看统计", CategoryAdmin, "查看平台统计数据"},
        {PermSessionsRead, "查看会话", CategoryAdmin, "查看用户 Focus Sessions"},
        {PermSettingsRead, "查看设置", CategoryAdmin, "查看系统设置"},
        {PermSettingsWrite, "修改设置", CategoryAdmin, "修改系统设置"},
        // Frontend
        {PermNotionSyncEnable, "启用 Notion 同步", CategoryFrontend, "使用 Notion 同步功能"},
        {PermFocusModeAccess, "使用 Focus Mode", CategoryFrontend, "使用专注计时功能"},
        {PermBulkOpsAccess, "批量操作", CategoryFrontend, "执行批量任务操作"},
        {PermExportAccess, "数据导出", CategoryFrontend, "导出任务和集合数据"},
    }

    for _, def := range permissionDefs {
        var p models.Permission
        result := tx.Where("code = ?", def.Code).FirstOrCreate(&p, models.Permission{
            Code:        def.Code,
            Name:        def.Name,
            Category:    def.Category,
            Description: def.Description,
        })
        if result.Error != nil {
            return result.Error
        }
    }
    return nil
}

func (s *SeedService) seedRoles(tx *gorm.DB) error {
    // Super Admin — all permissions
    superAdmin, err := s.findOrCreateRole(tx, "Super Admin", "超级管理员，拥有全部权限", true)
    if err != nil {
        return err
    }
    var allPerms []models.Permission
    tx.Find(&allPerms)
    tx.Model(&superAdmin).Association("Permissions").Replace(allPerms)

    // Operator — partial admin + all frontend features
    operator, err := s.findOrCreateRole(tx, "Operator", "操作员，部分管理权限", true)
    if err != nil {
        return err
    }
    var adminPerms []models.Permission
    tx.Where("category = ?", CategoryAdmin).Find(&adminPerms)
    // Operator: read all + write tasks/collections/users (no delete, no roles manage, no settings write)
    operatorPerms := []models.Permission{}
    for _, p := range adminPerms {
        if p.Code == PermTasksDelete || p.Code == PermCollectionsDelete ||
           p.Code == PermUsersDelete || p.Code == PermRolesWrite ||
           p.Code == PermSettingsWrite {
            continue
        }
        operatorPerms = append(operatorPerms, p)
    }
    var frontendPerms []models.Permission
    tx.Where("category = ?", CategoryFrontend).Find(&frontendPerms)
    operatorPerms = append(operatorPerms, frontendPerms...)
    tx.Model(&operator).Association("Permissions").Replace(operatorPerms)

    // Viewer — read-only + no frontend features
    viewer, err := s.findOrCreateRole(tx, "Viewer", "查看者，只读权限", true)
    if err != nil {
        return err
    }
    var readPerms []models.Permission
    tx.Where("code IN (?)", []string{
        PermTasksRead, PermCollectionsRead, PermUsersRead,
        PermRolesRead, PermStatsView, PermSessionsRead,
        PermSettingsRead,
    }).Find(&readPerms)
    tx.Model(&viewer).Association("Permissions").Replace(readPerms)

    return nil
}

func (s *SeedService) findOrCreateRole(tx *gorm.DB, name, desc string, isSystem bool) (models.Role, error) {
    var role models.Role
    result := tx.Where("name = ?", name).FirstOrCreate(&role, models.Role{
        Name:        name,
        Description: desc,
        IsSystem:    isSystem,
    })
    return role, result.Error
}
```
